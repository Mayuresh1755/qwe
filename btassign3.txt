pragma solidity ^0.8.0;
contract Bank{
    uint256 balance=0;
    address public accOwner;

    constructor(){
        accOwner=msg.sender;
    }
    //deposit
    function Deposit() public payable {
        require(accOwner == msg.sender,"you are not an account owner");
        require(msg.value >0, "amount should be greater than zero");
        balance+=msg.value;

    }
    //withdraw
    function withdraw() public payable {
        require(accOwner == msg.sender, "you are not an account owner");
        require(msg.value >0, "amount should be greater than zero");
        balance-=msg.value;

    }

    //showbalance
    function showbalance() public view/**no operations are gonna perfromed only view**/ returns(uint256){
        require(accOwner == msg.sender, "you are not an account owner");
        return balance;


    }



}
